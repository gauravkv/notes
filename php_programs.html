Question: Demonstrate polymorphism using a common method implemented differently in multiple classes.
Solution: 
<?php
abstract class Shape {
    abstract public function area();
}

class Square extends Shape {
    private $side;
    public function __construct($side) { $this->side = $side; }
    public function area() { return $this->side * $this->side; }
}

class Rectangle extends Shape {
    private $length, $width;
    public function __construct($length, $width) {
        $this->length = $length;
        $this->width = $width;
    }
    public function area() { return $this->length * $this->width; }
}

$shapes = [new Square(4), new Rectangle(5, 3)];

foreach ($shapes as $shape) {
    echo "Area: " . $shape->area() . "\n";
}
?>

Area: 16 Area: 15
