class Message {
    private String content;
    private boolean available = false;

    synchronized void produce(String msg) {
        content = msg;
        available = true;
        notify();
    }

    synchronized String consume() throws InterruptedException {
        while (!available)
            wait();
        available = false;
        return content;
    }
}

class WaitNotify {
    public static void main(String[] args) throws InterruptedException {
        Message m = new Message();

        Thread consumer = new Thread(() -> {
            try {
                System.out.println(m.consume());
            } catch (InterruptedException e) {
            }
        });

        Thread producer = new Thread(() -> m.produce("Hello from Producer"));

        consumer.start();
        Thread.sleep(200);
        producer.start();

        producer.join();
        consumer.join();
    }
}

Output

Hello from Producer

import java.util.List;
import java.util.ArrayList;

class GenericStack<T> {
    private List<T> stack = new ArrayList<>();

    void push(T item) {
        stack.add(item);
    }

    T pop() {
        return stack.remove(stack.size() - 1);
    }

    boolean isEmpty() {
        return stack.isEmpty();
    }

    public static void main(String[] args) {
        GenericStack<Integer> s = new GenericStack<>();
        s.push(10);
        s.push(20);
        s.push(30);

        while (!s.isEmpty())
            System.out.println(s.pop());
    }
}
